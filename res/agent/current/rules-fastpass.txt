# ------------------------------------------------------------------------------
# 5. EXECUTION ARCHITECTURE & RUNTIME RULES
# ------------------------------------------------------------------------------
[EXECUTION]
non_op_code         = use `invariant(cond, msg)` (`libag_invariant.mjs`); halts immediately (`throw new Error`) on failure
op_controlled_code  = `run_operation(ctx, steps)` executes `step.call(ctx, options)` inside `try/catch`
op_lifecycle        = `run_pre_op_check(ctx)` -> steps with `retry(ctx, err, attempt)` & `micro_pause.retry_backoff(attempt, ctx)` -> `run_self_check(ctx)`
cooperative_yield   = `micro_pause.yield(ctx, reason)` governed by profile modifiers (`+bg`: 10ms, `+fg`: 1ms, `-nolimits`: 0ms)
zero_gc_layout      = pack node records & state flags into flat `SharedArrayBuffer` / `Uint8Array` / `Int32Array` strides with `Atomics` accessors; avoid object/string slop in hot loops
bitwise_safety      = JS coerces `| & ^ << >>` to 32-bit signed int; restrict bitwise ops to `Uint8Array`/`Int32Array` views (`Atomics.or`, `Atomics.and`); use `BigUint64Array` + `Atomics.load`/`store` for 64-bit fields (`mtime`, `size`)
link_parlance       = use `referrer` (pointer node) and `referent` (backing target); encode state bits into node `Flags` byte (no state strings)
worker_isolation    = NEVER mix main-thread and worker-thread imports in one file; split across:
                      1) `libag_worker_op.mjs` (frozen `PROTOCOL_OP` + shared binary schemas)
                      2) `libag_AgentWorkerClient.mjs` (main-thread isolate only)
                      3) `libag_AgentWorkerRuntime.mjs` (worker-thread isolate only)
logging_tier_1      = `imut_log` / `stamp`: always-on immutable audit log (lifecycle ops, `os_executor`, `host_fingerprint()`); NEVER clutter with high-freq loops
logging_tier_2      = User / CLI log: streamed console / `chalkpack` controlled by `+trace`, `+debug`, `+verbose`, `+send_debug`

# ------------------------------------------------------------------------------
# 6. RESIDENT PROCESS SECURITY & HARDENING
# ------------------------------------------------------------------------------
[SECURITY]
null_proto_dicts    = NEVER use `{}` for dynamic key/path caches; ALWAYS use `Object.create(null)` or `new Map()`
frozen_constants    = wrap all exported enums, flag masks, and OS command maps in `Object.freeze({...})`
safe_option_read    = NEVER use `options.foo || default`; ALWAYS use `Object.hasOwn(options, 'foo')` or `arg_get_opt(options, 'foo', default)`
profile_gating      = support `+`/`-` modifiers (`+bg`, `+fg`, `+session`, `+gate`, `+send_debug`) with constant-time `timingSafeEqual` PIN/token checks

# ------------------------------------------------------------------------------
# 7. BEHAVIORAL & INVARIANT TESTING
# ------------------------------------------------------------------------------
[TESTING]
zero_expected_throws = do NOT write mock-heavy unit tests asserting thrown errors ("if low-level code throws, it is borked")
behavioral_harness   = run via native `node --test` (`npm test`) on real/simulated workflows; verify `SharedArrayBuffer` strides, `0xAA` canary bytes, state transitions, and manifests

[PATTERN:MODULE_TEMPLATE]
```javascript
// 1p
import { timingSafeEqual } from 'node:crypto'

// 2p
import { invariant } from '../internal/libag_invariant.mjs'
import { arg_get_opt } from '../arg/libag_arg.mjs'

// 3p
// (none - zero external npm dependencies in resident core)

export const MODULE_TEMPLATE_FLAGS= Object.freeze({
    cold: 0x01,
    active: 0x02,
    dirty: 0x04,
    resident: 0x08
})

export function template_init_registry(options= {}) {
    var opts= options || {}
    var max_entries= arg_get_opt(opts, 'max_entries', 1024)
    var profile= arg_get_opt(opts, 'profile', '+bg')

    return {
        cache: Object.create(null),
        registry: new Map(),
        state_flags: MODULE_TEMPLATE_FLAGS.cold,
        max_entries: max_entries,
        profile: profile
    }
}

export function template_state_flags(ctx, val) {
    invariant(ctx && typeof ctx === 'object', 'template_state_flags: valid ctx required')
    if (val !== undefined) {
        ctx.state_flags= val
        return val
    }
    return ctx.state_flags
}

export class libag_ModuleTemplate {
    #ctx

    constructor(options= {}) {
        this.#ctx= template_init_registry(options)
    }

    state_flags(val) {
        return template_state_flags(this.#ctx, val)
    }
}
```
