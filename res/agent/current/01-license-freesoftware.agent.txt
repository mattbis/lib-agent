target      :: *.mjs | *.zig | *.sh | *.cmd

path        :: `src-mjs.dev/3p/<pkg>/`
allow_lic   :: GPLv3_COMPAT_ONLY {GPL-3.0, LGPL-3.0, AGPL-3.0, Apache-2.0, MIT, BSD-2/3, ISC, MPL-2.0}
forbid_lic  :: PROPRIETARY | NON_FREE | SOURCE_AVAILABLE_NON_COMMERCIAL | GPLv3_INCOMPAT
