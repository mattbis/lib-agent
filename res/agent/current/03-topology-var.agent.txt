[AXIOM::VAR_OUTPUT_DISCIPLINE]
!no_root_splatter :: FORBID scattered `target/`, `build/`, `dist/`, `__pycache__/`, `.gradle/`, `zig-cache/` in repo root or `src-*`
var_build         :: `var/build/<lang-or-target>/` |> ALL compiler outputs, object caches, and intermediate bundles (`.gitignore`d)
var_dist          :: `var/dist/`                   |> package-manager distribution artifacts (`.gitignore`d)
var_release       :: `var/release/`                |> cryptographically signed release artifacts & manifests (`.gitignore`d)

[INVARIANTS::CHANNEL_HYGIENE]
!preserve_stubs   :: empty files in `src-<lang>.dev/` == intentional architectural placeholders |> NEVER delete
!script_ext       :: `local/sh/*` MUST use explicit `.sh` or `.cmd` extension
